cmake_minimum_required(VERSION 3.21)
project(TuringScreenQt VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)
add_library(screen_core STATIC src/model.cpp src/nrbf.cpp src/renderer.cpp src/sensors.cpp src/device.cpp)
target_include_directories(screen_core PUBLIC src)
target_link_libraries(screen_core PUBLIC Qt6::Core Qt6::Gui)
target_compile_definitions(screen_core PUBLIC SCREEN_VERSION="${PROJECT_VERSION}" SCREEN_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(MSVC)
    target_compile_options(screen_core PUBLIC /utf-8 /W4 /permissive-)
    target_link_libraries(screen_core PUBLIC iphlpapi powrprof advapi32 psapi ole32 uuid)
else()
    target_compile_options(screen_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(turing-screen-qt WIN32 src/main.cpp src/mainwindow.cpp src/canvas.cpp)
target_link_libraries(turing-screen-qt PRIVATE screen_core Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(screen-tests tests/core_test.cpp)
    target_link_libraries(screen-tests PRIVATE screen_core Qt6::Test)
    target_compile_definitions(screen-tests PRIVATE SCREEN_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME core COMMAND screen-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(gui-tests tests/gui_test.cpp src/mainwindow.cpp src/canvas.cpp)
    target_link_libraries(gui-tests PRIVATE screen_core Qt6::Widgets Qt6::Test)
    add_test(NAME gui COMMAND gui-tests)
    set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS turing-screen-qt RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
install(FILES README.md LICENSE THIRD_PARTY.md DESTINATION .)
