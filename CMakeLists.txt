cmake_minimum_required(VERSION 3.21)
project(TuringScreenQt VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(SCREEN_ASSET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/assets" CACHE PATH "Runtime asset fallback directory")
set(SCREEN_INSTALL_BINDIR "." CACHE STRING "Executable installation directory")
set(SCREEN_INSTALL_DATADIR "." CACHE STRING "Assets and documentation installation directory")
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Svg)
qt_add_executable(screen-icon-generator EXCLUDE_FROM_ALL tools/generate_app_icon.cpp)
target_link_libraries(screen-icon-generator PRIVATE Qt6::Gui Qt6::Svg)
add_library(screen_core STATIC src/model.cpp src/nrbf.cpp src/renderer.cpp src/sensors.cpp src/systeminfo.cpp src/memoryinfo.cpp src/linuxhardware.cpp src/windowshardware.cpp src/audio.cpp src/weather.cpp src/device.cpp src/runtime.cpp src/startup.cpp src/themelibrary.cpp)
target_include_directories(screen_core PUBLIC src)
target_link_libraries(screen_core PUBLIC Qt6::Core Qt6::Gui Qt6::Network)
target_compile_definitions(screen_core PUBLIC SCREEN_VERSION="${PROJECT_VERSION}" SCREEN_ASSET_DIR="${SCREEN_ASSET_DIR}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
    target_link_libraries(screen_core PRIVATE PkgConfig::PULSE)
endif()
if(MSVC)
    target_compile_options(screen_core PUBLIC /utf-8 /W4 /permissive-)
    target_link_libraries(screen_core PUBLIC iphlpapi advapi32 psapi ole32 uuid)
else()
    target_compile_options(screen_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
set(SCREEN_UI_SOURCES src/mainwindow.cpp src/canvas.cpp src/application.cpp src/uiappearance.cpp src/themelibraryui.cpp src/panesplitter.cpp)
qt_add_executable(turing-screen-qt WIN32 src/main.cpp ${SCREEN_UI_SOURCES})
if(WIN32)
    enable_language(RC)
    configure_file(src/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
    target_sources(turing-screen-qt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
endif()
target_link_libraries(turing-screen-qt PRIVATE screen_core Qt6::Widgets Qt6::Svg)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(screen-tests tests/core_test.cpp)
    target_link_libraries(screen-tests PRIVATE screen_core Qt6::Test)
    target_compile_definitions(screen-tests PRIVATE SCREEN_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    add_test(NAME core COMMAND screen-tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(gui-tests tests/gui_test.cpp ${SCREEN_UI_SOURCES})
    target_link_libraries(gui-tests PRIVATE screen_core Qt6::Widgets Qt6::Svg Qt6::Test)
    add_test(NAME gui COMMAND gui-tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/gui-test.txt,txt")
    set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(WIN32)
        add_test(NAME gui-native COMMAND gui-tests
            sidebarIconsStayAlignedAndPageHeadersHaveEqualHeight
            sensorColumnsResizeFromHeaderAndKeepTheirWidths
            libraryToolbarSearchAndTreeActions collapsiblePanesRestoreAcrossOrientationsAndWindowRebuild
            overviewHasAFileAndFolderOpenKeepsThumbnails
            trayMenuKeepsSystemStyleAcrossPalettes traySerialMenuListsPortsAndMarksOnlyConnectedPort
            -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/gui-native-test.txt,txt")
        set_tests_properties(gui-native PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows:fontengine=gdi")
    endif()
    qt_add_executable(startup-tests tests/startup_test.cpp)
    target_link_libraries(startup-tests PRIVATE screen_core Qt6::Test)
    add_test(NAME startup COMMAND startup-tests)
    qt_add_executable(windows-sensor-tests tests/windows_sensors_test.cpp)
    target_link_libraries(windows-sensor-tests PRIVATE screen_core Qt6::Test)
    add_test(NAME windows-sensors COMMAND windows-sensor-tests)
endif()
option(SCREEN_BUILD_PROFILING "Build the standalone native performance probe" OFF)
if(SCREEN_BUILD_PROFILING)
    qt_add_executable(screen-profile tools/performance_probe.cpp ${SCREEN_UI_SOURCES})
    target_link_libraries(screen-profile PRIVATE screen_core Qt6::Widgets Qt6::Svg)
endif()
install(TARGETS turing-screen-qt RUNTIME DESTINATION "${SCREEN_INSTALL_BINDIR}")
install(DIRECTORY assets DESTINATION "${SCREEN_INSTALL_DATADIR}")
install(DIRECTORY docs DESTINATION "${SCREEN_INSTALL_DATADIR}")
install(FILES README.md README.zh-CN.md LICENSE THIRD_PARTY.md THIRD_PARTY.zh-CN.md DESTINATION "${SCREEN_INSTALL_DATADIR}")
